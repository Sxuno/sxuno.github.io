2) # select plugin while true; do # ------------------------------- # SCAN PLUGINS # ------------------------------- mapfile -t PLUGIN_ARRAY < <(find "$PLUGIN_DIR" -mindepth 1 -maxdepth 1 -type d -exec basename {} \;) if [ ${#PLUGIN_ARRAY[@]} -eq 0 ]; then echo "No plugins found in $PLUGIN_DIR" read -n1 -p "Press any key to continue..." break # nothing to build, exit loop fi # ------------------------------- # DISPLAY OPTIONS # ------------------------------- echo "Available plugins:" echo for plugin in "${PLUGIN_ARRAY[@]}"; 3) exit 0 ;; esac
